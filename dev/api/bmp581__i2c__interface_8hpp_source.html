162 Error Read(uint8_t reg, std::span<uint8_t> out) noexcept {
163 if (out.empty() || (out.size() > kMaxBurst)) {
164 return Error::InvalidArgument;
165 }
166 const std::array<uint8_t, 1> pointer{reg};
167 if (port_.WriteRead(address_, pointer.data(), pointer.size(), out.data(), out.size())) {

174 Error WriteByte(uint8_t reg, uint8_t value) noexcept {
175 const std::array<uint8_t, 2> frame{reg, value};
176 return port_.Write(address_, frame.data(), frame.size()) ? Error::Ok : failure();
177 }

182 Error WriteResetCommand() noexcept {
183 const std::array<uint8_t, 2> frame{reg::kCmd, reg::kCmdSoftReset};
184 if (port_.Write(address_, frame.data(), frame.size())) {
185 return Error::Ok;
186 }
187 return (failure() == Error::NoDevice) ? Error::NoDevice : Error::Ok;
188 }

190private:
192 Error failure() noexcept {
193 if constexpr (HasGetLastFailure<Port>) {
194 I2cFailure reason = I2cFailure::Other;
195 if (port_.GetLastFailure(reason) && (reason == I2cFailure::AddressNack)) {
196 return Error::NoDevice;
197 }
