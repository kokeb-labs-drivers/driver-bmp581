60constexpr Oversampling OversamplingFromSamples(int32_t samples) noexcept {
61 uint32_t code = 0U;
62 for (uint32_t i = 1U; i < 8U; ++i) {
63 if (std::cmp_greater_equal(samples, uint32_t{1} << i)) {
64 code = i;
65 }
66 }
67 return static_cast<Oversampling>(code); // 0 to 7: every value is an enumerator
68}
