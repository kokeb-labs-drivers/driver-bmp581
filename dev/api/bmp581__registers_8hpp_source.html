134// DSP_IIR (section 7.28). The register map puts set_iir_t in bits 2..0 and set_iir_p in bits 5..3;
135// the field list of section 7.28 gives both "bit offset 0" and marks bits 5..3 read-only. The
136// driver follows the register map; the read-back after each write would show the other reading as
137// WriteMismatch (**not verified on hardware**).

164// The typed fields are the register map above, proved at build time (KB-CPP20-012).

166static_assert(IntConfigMode::Extract(kIntConfigReset) == 1U); // latched
167static_assert(IntConfigOd::Extract(kIntConfigReset) == 1U); // open drain
168static_assert(IntConfigEn::Extract(kIntConfigReset) == 0U); // pin disabled
169static_assert(IntConfigPadDrv::Extract(kIntConfigReset) == 3U);
170static_assert((FifoThreshold::kMask | FifoModeBit::kMask) == kFifoConfigMask);
171static_assert((FifoFrameSel::kMask | FifoDecSel::kMask) == kFifoSelMask);
172static_assert((DspIirT::kMask | DspIirP::kMask) == kDspIirMask);
173static_assert((OorThrP16::kMask | OorCntLim::kMask) == kOorConfigMask);
174static_assert(OdrOdr::Extract(kOdrConfigReset) == 0x1CU); // 1 Hz
175static_assert(OdrPwrMode::Extract(kOdrConfigReset) == 0U); // standby
176static_assert(DspCompPtEn::Extract(kDspConfigReset) == 3U);
177static_assert(OsrP::Insert(0x00_u8, 7_u32) == 0x38U);
178static_assert(OsrP::Extract(OsrP::Insert(0xFF_u8, 2_u32)) == 2U);
