139 Error SelectInterface() noexcept {
140 return Read(reg::kChipId, std::span<uint8_t>{rx_.data(), 1U});

145 Error Read(uint8_t reg, std::span<uint8_t> out) noexcept {
146 if (out.empty() || (out.size() > kSpiMaxBurst)) {
147 return Error::InvalidArgument;
148 }
149 const std::size_t count = out.size();
150 tx_.fill(0U);
151 tx_[0] = static_cast<uint8_t>(kSpiReadFlag | (static_cast<uint32_t>(reg) & kSpiAddressMask));
152 if (!port_.Transfer(tx_.data(), rx_.data(), count + 1U)) {
153 return Error::Bus;
154 }
155 for (std::size_t i = 0U; i < count; ++i) {
156 out[i] = rx_[1U + i];
157 }
158 return Error::Ok;
159 }
160
162 Error WriteByte(uint8_t reg, uint8_t value) noexcept {
163 const std::array<uint8_t, 2> frame{
164 static_cast<uint8_t>(static_cast<uint32_t>(reg) & kSpiAddressMask), value};
165 return port_.Transfer(frame.data(), nullptr, frame.size()) ? Error::Ok : Error::Bus;
166 }
167
169 Error WriteResetCommand() noexcept {
170 return WriteByte(reg::kCmd, reg::kCmdSoftReset);
171 }
172
173private:
174 Port& port_;
175 std::array<uint8_t, 1U + kSpiMaxBurst> tx_{};
176 std::array<uint8_t, 1U + kSpiMaxBurst> rx_{};
177};
