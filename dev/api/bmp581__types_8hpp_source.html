547constexpr uint32_t MaxOdrMilliHz(Oversampling pressure, Oversampling temperature) noexcept {
548 const uint32_t p = static_cast<uint32_t>(pressure) & 0x07U; // masked: 8 settings
549 const uint32_t t = static_cast<uint32_t>(temperature) & 0x07U;
550 return static_cast<uint32_t>(detail::kMaxOdrHz[p][t]) * 1000U;
551}

560constexpr uint32_t ConversionTimeMaxUs(Oversampling pressure, Oversampling temperature) noexcept {
561 const uint32_t typical = detail::kConvPressureUs[static_cast<uint32_t>(pressure) & 0x07U] +
562 detail::kConvTemperatureUs[static_cast<uint32_t>(temperature) & 0x07U];
563 return ((typical * 105U) + 99U) / 100U;
564}

642constexpr bool EnumeratorsValid(const Config& c) noexcept {
643 return IsEnumerator(c.mode) && IsEnumerator(c.pressure_oversampling) &&
644 IsEnumerator(c.temperature_oversampling) && IsEnumerator(c.odr) &&
645 IsEnumerator(c.pressure_iir) && IsEnumerator(c.temperature_iir) &&
646 IsEnumerator(c.fifo_frame) && IsEnumerator(c.fifo_mode) && IsEnumerator(c.int_mode) &&

654constexpr bool RangesValid(const Config& c) noexcept {
655 const uint32_t threshold_max = (c.fifo_frame == FifoFrame::PressureAndTemperature) ? 15U : 31U;
656 const bool fields =
657 (static_cast<uint32_t>(c.fifo_threshold) <= threshold_max) &&
658 (static_cast<uint32_t>(c.fifo_decimation) <= 7U) &&
659 (static_cast<uint32_t>(c.int_pad_drive) <= 15U) &&
660 ((c.int_drive == IntDrive::OpenDrain) || (static_cast<uint32_t>(c.int_pad_drive) <= 7U)) &&
661 (c.oor_reference_pa <= kOorReferenceMaxPa);
662 const bool timeouts = (c.startup_timeout_ms >= 1U) && (c.startup_timeout_ms <= kTimeoutMaxMs) &&
663 (c.data_ready_timeout_ms >= 1U) &&
664 (c.data_ready_timeout_ms <= kTimeoutMaxMs);
665 return fields && timeouts;
666}

681constexpr Error Validate(const Config& config) noexcept {
682 if (!detail::EnumeratorsValid(config) || (config.mode == PowerMode::Forced) ||
683 !detail::RangesValid(config)) {
684 return Error::InvalidArgument;
685 }
686 if ((config.mode == PowerMode::Normal) &&
687 (OdrMilliHz(config.odr) >
688 MaxOdrMilliHz(config.pressure_oversampling, config.temperature_oversampling))) {
689 return Error::InvalidArgument;
690 }
691 return Error::Ok;
692}
