125[[nodiscard]] constexpr uint8_t IntConfigValue(const Config& c) noexcept {
126 auto v = reg::IntConfigMode::Insert(0_u8, ToUnderlying(c.int_mode));
127 v = reg::IntConfigPol::Insert(v, ToUnderlying(c.int_polarity));
128 v = reg::IntConfigOd::Insert(v, ToUnderlying(c.int_drive));
129 v = reg::IntConfigEn::Insert(v, c.int_pin_enable ? 1_u32 : 0_u32);
130 return reg::IntConfigPadDrv::Insert(v, c.int_pad_drive);

134[[nodiscard]] constexpr uint8_t IntSourceValue(const Config& c) noexcept {
135 auto v = SetIf(0_u8, reg::kIntDrdy, c.source_data_ready);
136 v = SetIf(v, reg::kIntFifoFull, c.source_fifo_full);

294[[nodiscard]] constexpr Result<Measurement> DecodeMeasurement(uint32_t temperature,
295 uint32_t pressure) noexcept {
296 if ((temperature == reg::kDataReset) || (pressure == reg::kDataReset)) {
297 return Result<Measurement>::Failure(Error::NoData);
298 }
299 const Measurement m{.pressure_raw = SignExtend24(pressure),
300 .temperature_raw = SignExtend24(temperature)};
301 if (!PressureInRange(m.pressure_raw) || !TemperatureInRange(m.temperature_raw)) {
302 return Result<Measurement>::Failure(Error::OutOfRange);
303 }
304 return Result<Measurement>::Success(m);
305}

330[[nodiscard]] constexpr FifoSample DecodeFifoValue(FifoSample s, uint32_t raw,
331 bool pressure) noexcept {
332 if (s.error != Error::Ok) {
333 return s;
334 }
335 if (raw == reg::kDataReset) {
336 return FifoSample{.error = Error::NoData};
337 }
338 const int32_t value = SignExtend24(raw);
339 if (pressure ? !PressureInRange(value) : !TemperatureInRange(value)) {
340 return FifoSample{.error = Error::OutOfRange};
341 }
342 if (pressure) {
343 s.has_pressure = true;
344 s.pressure_raw = value;
345 } else {
346 s.has_temperature = true;
347 s.temperature_raw = value;
348 }
349 return s;
350}

353[[nodiscard]] constexpr FifoSample DecodeFifoFrame(FifoFrame frame,
354 std::span<const uint8_t, 6> b) noexcept {
355 const uint32_t first = Le24(b.first<3>());
356 if (frame == FifoFrame::PressureAndTemperature) {
357 const FifoSample t = DecodeFifoValue(FifoSample{}, first, false);
358 return DecodeFifoValue(t, Le24(b.last<3>()), true);
359 }
360 return DecodeFifoValue(FifoSample{}, first, frame == FifoFrame::Pressure);
361}

363// The reset configuration encodes to the reset values of section 7, except press_en.
364static_assert(IntConfigValue(Config{}) == reg::kIntConfigReset);
365static_assert(DspConfigValue(Config{}) == reg::kDspConfigReset);
366static_assert(OdrConfigValue(Config{}, PowerMode::Standby) == reg::kOdrConfigReset);
367static_assert(OsrConfigValue(Config{}) == reg::kOsrPressEn);
368static_assert((IntSourceValue(Config{}) == 0U) && (FifoConfigValue(Config{}) == 0U) &&
369 (FifoSelValue(Config{}) == 0U) && (DspIirValue(Config{}) == 0U) &&
370 (OorConfigValue(Config{}) == 0U));
371// No configuration encodes OSR_CONFIG without press_en; the reset value shows a reset.
372static_assert(!OsrShowsReset(OsrConfigValue(HighResolutionPreset())));
373static_assert(OsrShowsReset(reg::kOsrConfigReset));
374// The example of section 4.7.2.3: 97150 Pa +/- 50 Pa.
375static_assert(OorConfigValue(Config{.oor_reference_pa = 97'150U}) == 0x01U);
376static_assert(ConfigWrites(Config{.oor_reference_pa = 97'150U})[6].value == 0x7BU);
377static_assert(ConfigWrites(Config{.oor_reference_pa = 97'150U})[5].value == 0x7EU);
378// The FIFO frames of tables 11 to 13 and the empty frame of table 14.
379static_assert(Le24(std::array<uint8_t, 3>{0x00U, 0xA6U, 0x0EU}) == 0x0E'A600U);
380static_assert(DecodeFifoFrame(FifoFrame::Pressure,
381 std::array<uint8_t, 6>{0x00U, 0xA6U, 0x5EU, 0U, 0U, 0U})
382 .pressure_raw == 6'202'880);
383static_assert(DecodeFifoFrame(FifoFrame::Temperature,
384 std::array<uint8_t, 6>{0x7FU, 0x7FU, 0x7FU, 0U, 0U, 0U})
385 .error == Error::NoData);
386static_assert(DecodeMeasurement(0x7F'7F7FU, 0x7F'7F7FU).error == Error::NoData);
387static_assert(DecodeMeasurement(0x19'0000U, 0x61'A800U).value.pressure_raw == 6'400'000);
388static_assert(DecodeMeasurement(0U, 0U).error == Error::OutOfRange);
389static_assert(DecodeIntStatus(reg::kIntStatusAfterReset).power_on_reset);
